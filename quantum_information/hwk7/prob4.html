hbar = 1.0
alpha = 1.0

num_points = 100
t_list = np.linspace(0, t_max, num_points)

a = destroy(N)
adag = a.dag()

H0 = omega * (adag * a + 0.5)

x_op = (a + adag) / np.sqrt(2)
p_op = 1j * (adag - a) / np.sqrt(2)
