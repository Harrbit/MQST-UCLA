import numpy as np
import matplotlib.pyplot as plt

def true_P1(tw, Omega, delta, tp):
    Omega_prime = np.sqrt(Omega**2 + delta**2)
    # Break it down for readability:
    term1 = Omega_prime * np.cos(delta * tw / 2) * np.sin(Omega_prime * tp)
    term2 = 2 * delta * np.sin(delta * tw / 2) * (np.sin(Omega_prime * tp / 2))**2
    inside = (term1 - term2)**2
    prefactor = Omega**2 / (Omega_prime**4)
    
    return 1.0 - prefactor * inside
def measure_P1(tw, Omega, delta, tp, N):
    p_true = true_P1(tw, Omega, delta, tp)
    k = np.random.binomial(N, p_true)
    return k / N

Phi_values = np.linspace(-1.0, 1.0, 21)

N_values = [1, 10, 100, 1000]

plt.figure(figsize=(8,6))

for N in N_values:
    P1_measured = []
    for Phi in Phi_values:
        detuning = delta_nominal + gamma * Phi
        
        P1_meas = measure_P1(tw0, Omega, detuning, tp, N)
        P1_measured.append(P1_meas)
    
    plt.plot(Phi_values, P1_measured, 'o-', label=f'N = {N}')
